<script>
(function(){
  var root=document.getElementById('bhr-{{ section.id }}'); if(!root) return;
  var tabs=[].slice.call(root.querySelectorAll('.bhr__tab'));
  function show(t){tabs.forEach(function(x){var on=x===t;x.setAttribute('aria-selected',on);var p=document.getElementById(x.getAttribute('aria-controls'));if(p)p.hidden=!on;});}
  tabs.forEach(function(t,i){
    t.addEventListener('click',function(){show(t);});
    t.addEventListener('keydown',function(e){var k=e.key==='ArrowRight'?1:e.key==='ArrowLeft'?-1:0;if(!k)return;var n=tabs[(i+k+tabs.length)%tabs.length];n.focus();show(n);});
  });
})();
</script>
